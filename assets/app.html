🟢 开启实时捕获 Fiddler 推送模式：工具启动接收服务，Fiddler 负责抓包
Fiddler 推送模式：工具启动接收服务（9090），Fiddler 负责抓包
工作原理：本工具不直接抓包，而是接收 Fiddler 抓到的请求。浏览器 / App 仍然走 Fiddler 代理（默认 127.0.0.1:8888），由 Fiddler 通过 CustomRules 实时把每个完成的 session POST 到本工具。
配置步骤：
  1. 点击上方按钮，看到顶部状态变为「代理运行中」（接收服务 127.0.0.1:9090）。
  2. 打开 Fiddler，进入 Tools → Options → HTTPS，勾选 Capture HTTPS CONNECTs 与 Decrypt HTTPS traffic（装根证书时全点 Yes），否则只能抓到 https-tunnel 空记录。
  3. 进入 Rules → Customize Rules（快捷键 Ctrl+R）。注意 Fiddler 实际加载的是 C:\Users\PC\Documents\Fiddler2\Scripts\CustomRules.js，不是项目里的 fiddler/CustomRules.js。
  4. 把下方 Fiddler CustomRules 脚本 整段复制，粘贴到 CustomRules.js 里（替换掉任何旧的 OnDoneResponse 代码）。注意钩子名必须是 OnDone。
  5. 按 Ctrl+S 保存，Fiddler 自动重载脚本。若状态栏出现 FiddlerScript rules failed to load，说明脚本语法不兼容，需根据弹窗提示修复。
  6. 在浏览器 / App 中正常操作，命中的接口会自动出现在「抓包会话」页（仅白名单 host，默认 adtest.virtual.007ka.cn）。
  7. 要监控更多域：往 C:\Users\PC\Documents\Fiddler2\Scripts\captrace_hosts.txt 加一行 host 后缀，Ctrl+R 重载即生效。
  8. 排错：如果 9090 状态灯绿但会话没进来，打开 Fiddler 底部 Log（View → Show Debug Log），搜索 CAPTRACE_PUSH_ERR。常见原因是 Fiddler 的 WebClient 默认走系统代理，导致向 127.0.0.1:9090 推送时被 Fiddler 自身再次拦截；下方脚本已改用 HttpWebRequest + Proxy = null 绕过。
Fiddler CustomRules 脚本 
import System.Net;
import System.IO;

// ===== BEGIN: captrace push patch =====
// Fiddler 官方支持的"会话结束"钩子是 OnDone（OnDoneResponse 会被 Fiddler 静默忽略）。
// 作用：会话处理完毕后，把关键字段以 JSON 推送到助手的本地接收器
//       http://127.0.0.1:9090/ingest，由助手落入会话池。
//
// 【过滤策略】只推送"用户在 Fiddler Filters 里留下的"业务 session：
//   1) HTTPS 未解密的 CONNECT/tunnel 直接不推（url 会变成 https-tunnel://，没有意义）
//   2) Fiddler UI 隐藏的 session（用户在 session 列表右键 "Hide"）直接不推
//   3) host 命中自定义白名单文件 captrace_hosts.txt 时直接推（每行一个 host 后缀，
//      写在 Fiddler 脚本同级目录 C:\Users\PC\Documents\Fiddler2\Scripts\）
//   4) 未命中白名单但命中内置业务域（adtest.virtual.007ka.cn）也推
//
// 注意：本补丁使用 OnDone 钩子（会话已处理完毕），GetResponseBodyAsString() 在此时是安全的，
//       不会出现"无效 GZip 幻数"问题（那是 OnDoneResponse 时机的陷阱）。所以这里一并取响应体，
//       让助手能拿到完整返回值，用于和数据库做业务字段对比。
//
// 【踩坑】Fiddler.WebFormats.JSON.JsonEncode 在 JScript.NET 里偶尔对带 null 字段的对象
//        返回 null，导致 UploadString 抛"值不能为 null"；改为手拼 JSON 字符串。
// 【踩坑】Fiddler 里的 System.Net.WebClient 默认会走系统代理（也就是 Fiddler 自己），
//        向 127.0.0.1:9090 推送时会被 Fiddler 再次拦截，连接失败且 e.Message 为空
//        （Log 显示 CAPTRACE_PUSH_ERR undefined）。改用 HttpWebRequest + Proxy = null 可绕过。
static function _captrace_ShouldPush(oSession: Session): Boolean {
    if (oSession.isTunnel) return false;
    if (oSession.RequestMethod == 'CONNECT') return false;
    if (oSession.oFlags != null && oSession.oFlags.ContainsKey('ui-hide')) return false;

    var host = (oSession.host == null) ? '' : oSession.host.ToLower();
    var builtinWhitelist = [
        'adtest.virtual.007ka.cn'
    ];
    var matchedBuiltin = false;
    for (var i:int = 0; i < builtinWhitelist.length; i++) {
        if (host == builtinWhitelist[i] || host.EndsWith('.' + builtinWhitelist[i])) {
            matchedBuiltin = true;
            break;
        }
    }

    var customMatched = false;
    var userListPath = 'C:\\Users\\PC\\Documents\\Fiddler2\\Scripts\\captrace_hosts.txt';
    if (System.IO.File.Exists(userListPath)) {
        try {
            var lines = System.IO.File.ReadAllLines(userListPath);
            for (var j:int = 0; j < lines.length; j++) {
                var line = lines[j];
                if (line == null) continue;
                line = line.Trim().ToLower();
                if (line.length == 0 || line.StartsWith('#')) continue;
                if (host == line || host.EndsWith('.' + line)) {
                    customMatched = true;
                    break;
                }
            }
        } catch(e) { /* 忽略读错误 */ }
    }

    return matchedBuiltin || customMatched;
}

static function OnDone(oSession: Session) {
    try { FiddlerApplication.Log.LogString("CAPTRACE_FIRED id=" + oSession.id + " host=" + oSession.host + " tunnel=" + oSession.isTunnel + " enc=" + oSession.oResponse.headers["Content-Encoding"]); } catch(_e) {}
    if (!_captrace_ShouldPush(oSession)) return;

    var url = oSession.fullUrl;
    var requestBody = '';
    try { requestBody = oSession.GetRequestBodyAsString(); } catch(e) { requestBody = ''; }
    var reqHeaders = '';
    try { reqHeaders = oSession.oRequest.headers.ToString(); } catch(e) { reqHeaders = ''; }
    var respHeaders = '';
    try { respHeaders = oSession.oResponse.headers.ToString(); } catch(e) { respHeaders = ''; }
    var responseBody = '';
    try {
        // gzip/deflate 响应需在 OnDone 先解压再读，否则 GetResponseBodyAsString() 会抛
        // "GZip 头中的幻数不正确"异常并阻塞 Fiddler。
        var ce = '';
        try { ce = oSession.oResponse.headers['Content-Encoding']; } catch(e) { ce = ''; }
        if (ce != null && (ce.ToLower().IndexOf('gzip') >= 0 || ce.ToLower().IndexOf('deflate') >= 0)) {
            try { oSession.utilDecodeResponse(); } catch(e2) { /* 解压失败则跳过 */ }
        }
        responseBody = oSession.GetResponseBodyAsString();
    } catch(e) { responseBody = ''; }
    var traceId = '';
    if (oSession.oRequest.headers.Exists('x-trace-id')) {
        traceId = oSession.oRequest.headers['x-trace-id'];
    } else if (oSession.oRequest.headers.Exists('trace-id')) {
        traceId = oSession.oRequest.headers['trace-id'];
    }
    var statusCode = (oSession.responseCode == null) ? 0 : oSession.responseCode;

    function jsonEscape(s: String): String {
        if (s == null) return '';
        // JScript.NET 对正则字面量支持不稳定，用逐字符替换最稳妥
        var sb = new System.Text.StringBuilder();
        for (var k:int = 0; k < s.length; k++) {
            var ch = s.charAt(k);
            if (ch == '\\') sb.Append('\\\\');
            else if (ch == '"') sb.Append('\\"');
            else if (ch == '\r') sb.Append('\\r');
            else if (ch == '\n') sb.Append('\\n');
            else if (ch == '\t') sb.Append('\\t');
            else sb.Append(ch);
        }
        return sb.ToString();
    }
    var payload = '{'
        + '"id":' + '"' + jsonEscape(oSession.id.toString()) + '",'
        + '"method":' + '"' + jsonEscape(oSession.RequestMethod) + '",'
        + '"url":' + '"' + jsonEscape(url) + '",'
        + '"host":' + '"' + jsonEscape(oSession.host) + '",'
        + '"path":' + '"' + jsonEscape(oSession.PathAndQuery) + '",'
        + '"status_code":' + statusCode + ','
        + '"request_body":' + '"' + jsonEscape(requestBody) + '",'
        + '"request_headers":' + '"' + jsonEscape(reqHeaders) + '",'
        + '"response_headers":' + '"' + jsonEscape(respHeaders) + '",'
        + '"response_body":' + '"' + jsonEscape(responseBody) + '",'
        + '"trace_id":' + '"' + jsonEscape(traceId) + '"'
        + '}';

    try {
        var req = System.Net.WebRequest.Create('http://127.0.0.1:9090/ingest');
        req.Method = 'POST';
        req.ContentType = 'application/json';
        req.Proxy = null;
        var bytes = System.Text.Encoding.UTF8.GetBytes(payload);
        var stream = req.GetRequestStream();
        stream.Write(bytes, 0, bytes.Length);
        stream.Close();
        var resp = req.GetResponse();
        resp.Close();
        try { FiddlerApplication.Log.LogString("CAPTRACE_PUSH_OK id=" + oSession.id + " len=" + payload.Length); } catch(_e3) {}
    } catch(e) {
        var errType = '';
        var errMsg = '';
        try { errType = e.GetType().FullName; } catch(_t) {}
        try { errMsg = e.Message; } catch(_m) {}
        try { FiddlerApplication.Log.LogString("CAPTRACE_PUSH_ERR type=" + errType + " msg=" + errMsg); } catch(_e2) {}
    }
}
// ===== END: captrace push patch =====
